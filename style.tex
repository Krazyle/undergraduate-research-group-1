% LaTeX-level rules that Quarto does not expose as YAML keys.

% Paragraph indentation, including the first paragraph after a heading.
\usepackage{indentfirst}
\setlength{\parindent}{0.5in}

% Page numbers, outer top.
\usepackage{scrlayer-scrpage}
\clearpairofpagestyles
\ohead{\pagemark}
% Starred version also numbers chapter-opening pages, which otherwise lose
% their number because LaTeX uses the plain style there. Delete to match
% the old submission exactly.
\ohead*{\pagemark}

% Chapter numbers are typed by hand in each chapter heading, so KOMA's
% automatic number is switched off. Section numbers stay automatic.
\renewcommand*{\chapterformat}{}
\renewcommand*{\raggedchapter}{\centering}

% Quarto has no YAML key for citation and URL link colours, so set all of them
% here. Links stay clickable, they just stop printing in blue.
\AtBeginDocument{\hypersetup{colorlinks=true, allcolors=black, allbordercolors=black}}

% KOMA's default heading sizes and skips are not multiples of \baselineskip,
% so they break the uniform grid. Keep headings body-size and their skips at
% zero, so every line in the document advances by exactly one line.
\RedeclareSectionCommands[
  beforeskip=0pt,
  afterskip=1pt
]{chapter,section,subsection,subsubsection,paragraph}
\setkomafont{chapter}{\normalsize}
\setkomafont{section}{\normalsize}
\setkomafont{subsection}{\normalsize}
\setkomafont{subsubsection}{\normalsize}
\setkomafont{paragraph}{\normalsize}
\setkomafont{subparagraph}{\normalsize}

% Lists must not add their own vertical space.
\setlength{\parskip}{0pt}
\usepackage{enumitem}
\setlist{nosep}

\usepackage{etoolbox}
% Tables run one size down, so their columns hold more characters per line.
% The space before and after a table equals one double-spaced blank line. It
% is set before \footnotesize so it uses the body line height.
\AtBeginEnvironment{longtable}{\setlength{\LTpre}{\baselineskip}%
  \setlength{\LTpost}{\baselineskip}\footnotesize}

% Tables are single-spaced inside, as APA 7 allows, with clear room between rows.
\AtBeginEnvironment{longtable}{\setstretch{1}\renewcommand{\arraystretch}{1.8}}

% APA 7 captions for tables and figures: bold number on its own line, italic
% title below it, both flush left above the table or figure.
\AtBeginDocument{\captionsetup[table]{format=plain, labelfont=bf, textfont=it,
  labelsep=newline, justification=raggedright, singlelinecheck=false,
  font={normalsize,stretch=2}, skip=6pt}%
  \captionsetup[figure]{format=plain, labelfont=bf, textfont=it,
  labelsep=newline, justification=raggedright, singlelinecheck=false,
  font={normalsize,stretch=2}, skip=6pt}}

% Space before and after a figure also equals one double-spaced blank line.
\AtBeginEnvironment{figure}{\setlength{\intextsep}{\baselineskip}}

% Display equations sit on the double-spaced grid like a line of text. LaTeX
% normally adds extra space above and below them, which doubles the gap.
\everydisplay\expandafter{\the\everydisplay
  \abovedisplayskip=0pt\belowdisplayskip=0pt
  \abovedisplayshortskip=0pt\belowdisplayshortskip=0pt}
