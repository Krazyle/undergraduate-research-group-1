\documentclass[tikz,border=1pt]{standalone}
\usepackage{fontspec}
\setmainfont{Times New Roman}

\usetikzlibrary{positioning,arrows.meta,fit}

% Pure black-and-white palette for print
\begin{document}
\begin{tikzpicture}[
    font=\small,
    >=Stealth,
    block/.style={draw=black, rounded corners=5pt, align=left,
        text width=3.9cm, minimum height=5.2cm, inner sep=8pt,
        font=\footnotesize, line width=0.9pt},
    model/.style={draw=black, rounded corners=5pt, align=left,
        text width=5.0cm, inner sep=8pt, font=\footnotesize,
        line width=0.9pt},
    tier/.style={draw=black, rounded corners=5pt, align=left,
        text width=3.9cm, minimum height=2.4cm, inner sep=8pt,
        font=\footnotesize, line width=0.9pt},
    group/.style={draw=black, dashed, rounded corners=8pt,
        inner xsep=10pt, inner ysep=22pt, line width=0.6pt},
    grouplabel/.style={font=\footnotesize\bfseries, inner sep=6pt},
    twoway/.style={<->, very thick},
    bigarrow/.style={->, very thick},
    lbl/.style={font=\footnotesize\itshape, align=center}
]

    \node[block] (data) {
        \centerline{\small\bfseries Data Preparation}\\[8pt]
        $\bullet$~Collect tweets through\\
        \hspace*{0.9em}the X API\\[6pt]
        $\bullet$~Clean the tweets and\\
        \hspace*{0.9em}keep Taglish tweets\\[6pt]
        $\bullet$~Label both tasks with\\
        \hspace*{0.9em}two annotators\\[6pt]
        $\bullet$~Split into training,\\
        \hspace*{0.9em}validation, and test sets
    };

    \node[block, right=1.6cm of data] (train) {
        \centerline{\small\bfseries Training and Measurement}\\[8pt]
        $\bullet$~Choose the settings of\\
        \hspace*{0.9em}each model with seed 0\\[6pt]
        $\bullet$~Train 15 models with\\
        \hspace*{0.9em}3 seeds (45 main runs)\\[6pt]
        $\bullet$~Record the training time\\
        \hspace*{0.9em}and peak memory\\[6pt]
        $\bullet$~Save the best epoch\\
        \hspace*{0.9em}of each run
    };

    \node[block, right=1.6cm of train] (eval) {
        \centerline{\small\bfseries Evaluation and Analysis}\\[8pt]
        $\bullet$~Score the saved models\\
        \hspace*{0.9em}on the test set\\[6pt]
        $\bullet$~Compute F1-scores and\\
        \hspace*{0.9em}negative transfer\\[6pt]
        $\bullet$~Run the two-way ANOVA\\
        \hspace*{0.9em}and Tukey's HSD test\\[6pt]
        $\bullet$~Build the Pareto frontier
    };

    \node[model, above=1.3cm of train] (mtm) {
        \centerline{\small\bfseries Multi-Task Model}\\[8pt]
        $\bullet$~Shared encoder,\\
        \hspace*{0.9em}SEA-LION-ModernBERT-300M\\[6pt]
        $\bullet$~Classification head and NER head\\[6pt]
        $\bullet$~One of the 13 multi-task optimizers
    };

    \draw[bigarrow] (data.east) -- node[lbl, above=2pt] {labeled\\dataset} (train.west);
    \draw[bigarrow] (train.east) -- node[lbl, above=2pt] {saved\\models\\and logs} (eval.west);
    \draw[bigarrow] (mtm.south) -- node[lbl, right=2pt] {model to train} (train.north);

    % Prototype web application: three tiers below the benchmarking system
    \node[tier, below=3.0cm of data] (ui) {
        \centerline{\small\bfseries Interface}\\[6pt]
        $\bullet$~React, in a web browser\\[4pt]
        $\bullet$~Shows posts, filters,\\
        \hspace*{0.9em}summaries, and forms
    };

    \node[tier, below=3.0cm of train] (server) {
        \centerline{\small\bfseries Server}\\[6pt]
        $\bullet$~FastAPI in Python\\[4pt]
        $\bullet$~Runs the trained model\\
        \hspace*{0.9em}with PyTorch
    };

    \node[tier, below=3.0cm of eval] (db) {
        \centerline{\small\bfseries Database}\\[6pt]
        $\bullet$~PostgreSQL\\[4pt]
        $\bullet$~Stores posts, labels,\\
        \hspace*{0.9em}and statuses
    };

    \draw[twoway] (ui.east) -- node[lbl, above=2pt] {requests} node[lbl, below=2pt] {results} (server.west);
    \draw[twoway] (server.east) -- node[lbl, above=2pt] {queries} node[lbl, below=2pt] {records} (db.west);
    \draw[bigarrow] (train.south) -- node[lbl, right=2pt] {trained multi-task model} (server.north);

    \node[group, fit=(mtm) (data) (train) (eval)] (bench) {};
    \node[grouplabel, anchor=north west] at (bench.north west) {Part 1: Benchmarking System};
    \node[group, fit=(ui) (server) (db)] (proto) {};
    \node[grouplabel, anchor=north west] at (proto.north west) {Part 2: Prototype Web Application};

\end{tikzpicture}
\end{document}
